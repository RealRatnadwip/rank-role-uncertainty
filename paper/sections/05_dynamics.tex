\section{Markovian Attention Drift and Regime Transitions}
\label{sec:dynamics}

Even prior to a direct status inquiry, a multi-partner portfolio rarely remains static.
Over days and weeks, conversational context shifts across routine weekday workdays,
weekend leisure, and transient emotional stress. This section models temporal attention
drift using finite-state Markov chains~\citep{kemeny1960finite,levin2017markov,ross1996stochastic},
contrasting a time-homogeneous $12$-state augmented chain against a periodic weekly
stroboscopic operator, isolating absorbing post-collapse states from irreducible pre-poke
regimes, and showing how dynamical occupancy diverges from structural role dominance.

\subsection{Regime Transition Kernels and Homogeneous Augmented Representation}
\label{subsec:dyn-regimes}

\noindent\layerformal{} Index the active partner slots by $\{1, \dots, n\}$ and let
$\mathcal{E} = \{\mathrm{wd}, \mathrm{we}, \mathrm{st}\}$ denote the three behavioral
contexts: \emph{weekday routine} ($\mathrm{wd}$), \emph{weekend leisure} ($\mathrm{we}$),
and \emph{emotional stress} ($\mathrm{st}$). Under each regime $r \in \mathcal{E}$,
attention moves across partners according to a row-stochastic transition kernel $P^{(r)}
\in [0, 1]^{n \times n}$ satisfying $P_{ij}^{(r)} \ge 0$ and $\sum_{j=1}^n P_{ij}^{(r)} = 1$.

At the behavioral level (\layerbehav{}), the regime process itself evolves cyclically across
weekdays (5 days) and weekends (2 days) with exogenous stress shocks occurring independently
with daily probability $p_{\mathrm{st}} \in (0, 1)$. Rather than leaving this as an informal
time-inhomogeneous loop, we formalize the exact \emph{homogeneous augmented Markov chain}
on the product state space $\mathcal{S} = \partnerset \times \mathcal{E}$ (dimension $3n = 12$
when $n = 4$):
\begin{equation}
  P_{\mathrm{aug}}\bigl((i, r) \to (j, r')\bigr)
  = Q_{\mathrm{reg}}(r \to r') \, P_{ij}^{(r')},
  \label{eq:daily-kernels}
\end{equation}
where $Q_{\mathrm{reg}}$ governs daily regime transitions (with $\mathrm{wd} \to \mathrm{we}$
rate $1/5$, $\mathrm{we} \to \mathrm{wd}$ rate $1/2$, and stress entry probability
$p_{\mathrm{st}} = 0.15$). Notice that $P_{\mathrm{aug}}$ is verified row-stochastic
to machine precision in \texttt{tests/test\_rst\_invariants.py}, and convex regime mixtures
of row-stochastic matrices are proved row-stochastic in \texttt{RST.Dynamics.Markov}.

\subsection{Stroboscopic Weekly Operator, Ergodicity, and Absorbing States}
\label{subsec:dyn-stationary}

If deterministic weekly periodicity is sampled once every seven days (in the stress-free
idealization $p_{\mathrm{st}} = 0$), partner evolution over full weekly blocks obeys the
\emph{stroboscopic weekly transition matrix}
\begin{equation}
  P_{\mathrm{week}} = \bigl(P^{(\mathrm{wd})}\bigr)^5 \bigl(P^{(\mathrm{we})}\bigr)^2.
  \label{eq:weekly-kernel}
\end{equation}

\begin{proposition}[Stationary attention under irreducible aperiodic drift; \layermath]
\label{prop:stationary}
If a finite row-stochastic matrix $P$ (either $P_{\mathrm{week}}$ or $P_{\mathrm{aug}}$)
is strictly positive ($P_{ij} > 0$ for all $i, j$), then $P$ is irreducible and aperiodic
and possesses a unique strictly positive stationary distribution $\pi$ ($\pi^\top P = \pi^\top$),
with geometric convergence in total variation (\cref{app:proof-doeblin}). Consequently,
for almost every sample trajectory with smoothing $\alpha \in (0, 1)$, time-averaged
occupancy approaches $\pi$:
\begin{equation}
  \lim_{T \to \infty} \frac{1}{T} \sum_{t=0}^{T-1} \mathbf{1}\{S_t = j\} = \pi_j > 0.
  \label{eq:ergodic-limit}
\end{equation}
\end{proposition}
\begin{proof}
Strict positivity gives $P_{ij} \ge \varepsilon_0 > 0$, making all states mutually
communicating in one step (irreducible and aperiodic). Uniqueness and exponential
total-variation mixing follow immediately from Doeblin's minorization theorem
(\cref{lem:app-doeblin}) and the ergodic theorem for finite chains~\citep{levin2017markov}.
\end{proof}

\noindent\layerformal{} In stark contrast, a monomorphic post-collapse regime corresponds to
an \emph{absorbing state} $k$ satisfying $P_{kk} = 1$ (hence $P_{kj} = 0$ for all $j \neq k$).
As formalized in \texttt{Dynamics.\allowbreak{}StateTransitions} (\texttt{absorbing\_\allowbreak{}point\_mass\_invariant}),
once weight concentrates onto $\delta_k$, Markov propagation leaves $\delta_k$ invariant
forever, destroying irreducibility. Thus, pre-poke indefinite persistence of superposition
depends mathematically on preserving non-zero off-diagonal transition probabilities across
partners.

To translate discrete conversational steps $S_t$ into continuous smoothed weights
$\mathbf{w}(t) \in \Delta^{n-1}$, \RST{} applies an exponential moving average (EMA):
\begin{equation}
  w_i(t + 1) = (1 - \alpha)\, w_i(t) + \alpha\, \mathbf{1}\{S_t = i\},
  \qquad \alpha \in (0, 1),
  \label{eq:ema-weights}
\end{equation}
which preserves $\sum_{i=1}^n w_i(t) = 1$ and $w_i(t) \ge 0$ at every step
(\texttt{emaUpdate} in \texttt{RST.Dynamics.\allowbreak{}StateTransitions}).

\subsection{Time-Averaged Mean, Variance, and Divergence from Role Dominance}
\label{subsec:dyn-variance}

\noindent\layermath{} For a time-homogeneous stationary Markov sequence $(S_t)$ with
stationary distribution $\pi$, the EMA weights satisfy $\E[w_i(t)] \to \pi_i$ as $t \to \infty$,
and in the i.i.d.\ approximation (\citealp{ross1996stochastic}) the asymptotic marginal
variance of partner $i$'s weight obeys
\begin{equation}
  \lim_{t \to \infty} \operatorname{Var}\bigl(w_i(t)\bigr)
  = \frac{\alpha}{2 - \alpha}\,\pi_i(1 - \pi_i).
  \label{eq:ema-variance}
\end{equation}

\begin{example}[Two-partner regime divergence; \layerformal]
\label{ex:markov-2x2}
Take two partners with weekday kernel $P^{(\mathrm{wd})} = \begin{psmallmatrix} 0.8 & 0.2 \\ 0.4 & 0.6 \end{psmallmatrix}$
and weekend kernel $P^{(\mathrm{we})} = \begin{psmallmatrix} 0.3 & 0.7 \\ 0.2 & 0.8 \end{psmallmatrix}$.
Individually, $\pi^{(\mathrm{wd})} = (2/3, 1/3)$ favors $G_1$, while
$\pi^{(\mathrm{we})} = (2/9, 7/9)$ favors $G_2$. Neither kernel alone describes the cycle;
composing weekly dynamics shifts occupancy smoothly without fixing an invariant ordinal rank.
\end{example}

\noindent\layerformal{} Crucially, Experiment~2 (\cref{sec:simulations}) demonstrates a deep
dissociation between \emph{dynamical attention occupancy} and \emph{static role importance}:
under the periodic stroboscopic kernel~\eqref{eq:weekly-kernel}, weekend inertia pushes
$\pi_{\mathrm{week}}(G_4) = 0.3779$ above $\pi_{\mathrm{week}}(G_1) = 0.2511$ and
$\pi_{\mathrm{week}}(G_2) = 0.2248$, even though $G_4$ scores dead last ($0.1458$) under the
support weighting scheme of \cref{ex:role-aggregation}; conversely, under the $12$-state
homogeneous joint chain with stress interruptions ($p_{\mathrm{st}} = 0.15$), attention
marginalizes to $(0.3334, 0.3087, 0.1741, 0.1838)$, placing $G_1$ and $G_2$ ahead of $G_4$.

\subsection{Metastability and Empirical Horizon}
\label{subsec:dyn-metastability}

\noindent\layermath{} When $P_{\mathrm{week}}$ has spectral gap $1 - \lambda_* > 0$
(second eigenvalue modulus $\lambda_* < 1$), total variation distance decays geometrically as
\begin{equation}
  d_{\mathrm{TV}}\bigl(\mu(0)^\top P_{\mathrm{week}}^k, \pi_{\mathrm{week}}\bigr)
  \;\le\; C\,\lambda_*^k.
  \label{eq:mixing-bound}
\end{equation}
On a $90$-day synthetic horizon (\cref{fig:markov-drift}), EMA weights fluctuate persistently
across $(0.10, 0.50)$ while Shannon entropy remains near $1.45$--$1.85$~bits, confirming
that temporal regime switching prevents spontaneous convergence to a singleton even when no
partner holds equal weight.
