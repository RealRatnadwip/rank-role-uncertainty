\section{Rank and Role Observables and the Uncertainty Inequality}
\label{sec:uncertainty}

This section models ``where do I stand?'' (rank) and ``what am I to you?'' (role) as
Hermitian operators on a finite-dimensional complex inner-product space and derives
the uncertainty bounds relating their dispersions. All operator identities and
inequalities in this section are classical finite-dimensional Hilbert-space mathematics
due to \citet{robertson1929uncertainty}, \citet{schrodinger1930unscharfeprinzip}, and
\citet{vonneumann1955mathematical}; \RST{}'s contribution is the behavioral interpretation
and explicit matrix construction relating diagonal rank to unitarily rotated role context.

\subsection{Rank and Role as Hermitian Observables}
\label{subsec:unc-operators}

\noindent\layermath{} Let $\mathcal{H} \cong \Cnum^d$ be a $d$-dimensional complex
inner-product space ($d \ge 2$) equipped with the standard inner product linear in the
second slot:
\begin{equation}
  \braket{u}{v} = \sum_{i=1}^{d} u_i^* v_i,
  \qquad \norm{u} = \sqrt{\braket{u}{u}}.
  \label{eq:inner-product}
\end{equation}
A linear operator $\hat{A} : \mathcal{H} \to \mathcal{H}$ is \emph{Hermitian}
($\hat{A}^\dagger = \hat{A}$) if $\braket{u}{\hat{A} v} = \braket{\hat{A} u}{v}$ for all
$u, v \in \mathcal{H}$. Real-valued spectra guarantee real measurement expectations and
real outcome averages.

\begin{definition}[Expectation, variance, and dispersion; \layermath]
\label{def:dispersion}
For any Hermitian observable $\hat{A}$ and unit state vector $\ket{\Psi} \in \mathcal{H}$
($\norm{\Psi} = 1$), the expectation value, centered operator $\hat{A}'$, variance, and
\emph{dispersion} (standard deviation) are
\begin{align*}
  \expect{\hat{A}}{\Psi} &= \braket{\Psi}{\hat{A}\Psi} \in \Rnum, &
  \hat{A}' &= \hat{A} - \expect{\hat{A}}{\Psi}\opI, \\
  \var{\hat{A}}{\Psi} &= \norm{\hat{A}'\Psi}^2 = \expect{(\hat{A}')^2}{\Psi}, &
  \Delta_{\Psi} A &= \sqrt{\var{\hat{A}}{\Psi}}.
\end{align*}
\end{definition}

When context is fixed, we write $\Delta R = \Delta_{\Psi} R$ and $\Delta L = \Delta_{\Psi} L$.
Notice that $\Delta_{\Psi} A = 0$ if and only if $\hat{A}\ket{\Psi} = \expect{\hat{A}}{\Psi}\ket{\Psi}$,
i.e., when $\ket{\Psi}$ is an exact eigenstate with zero outcome uncertainty.

\noindent\layerformal{} In the canonical diagonal rank basis $\{\ket{e_1}, \dots, \ket{e_d}\}$,
the idealized \emph{rank operator} $\opR = D_R = \diag(1, 2, \dots, d)$ assigns eigenvalue $k$
to slot $k$. Similarly, a canonical role evaluation matrix $D_L$ in its native context
assigns numerical role scores (for instance $\diag(-1, \dots, +1)$ from contingency to
primary visibility). As formalized in \cref{prop:diag-commute}, two operators diagonal in
the same basis always commute. Non-commutativity enters specifically when role interpretation
is evaluated relative to a rotated conversational context $U(\gamma)$:
\begin{equation}
  \opR = D_R, \qquad
  \opL(\gamma) = U(\gamma)^\dagger D_L U(\gamma),
  \label{eq:rotated-role}
\end{equation}
where $U(\gamma)$ is unitary ($U(\gamma)^\dagger U(\gamma) = I_d$) with $U(0) = I_d$.
Because matrix conjugation preserves Hermiticity (\texttt{conjugateBy\_\allowbreak{}hermitian} in
\texttt{Uncertainty.Commutator}), $\opL(\gamma)$ remains a valid real-spectrum observable
for every $\gamma$.

\subsection{The Commutator, Anticommutator, and Basis Alignment}
\label{subsec:unc-commutator}

\begin{definition}[Commutator and centered anticommutator; \layermath]
\label{def:commutator}
For two linear operators $\opR, \opL$ on $\mathcal{H}$, the \emph{commutator} and the
\emph{centered anticommutator} (with respect to state $\ket{\Psi}$) are
\begin{equation}
  \comm{\opR}{\opL} = \opR\opL - \opL\opR,
  \qquad
  \{\opR', \opL'\} = \opR'\opL' + \opL'\opR'.
  \label{eq:commutator}
\end{equation}
\end{definition}

\begin{proposition}[Diagonal operators commute identically; \layermath]
\label{prop:diag-commute}
If $\opR = \diag(a_1, \dots, a_d)$ and $\opL = \diag(b_1, \dots, b_d)$ in the same basis,
then $\comm{\opR}{\opL} = 0$. Consequently, at zero context rotation ($\gamma = 0$),
$\opL(0) = D_L$ commutes with $\opR = D_R$ and the commutator uncertainty bound vanishes.
\end{proposition}
\begin{proof}
Componentwise, $(D_a D_b \Psi)_i = a_i b_i \Psi_i = b_i a_i \Psi_i = (D_b D_a \Psi)_i$,
verified formally in \texttt{diagMat\_commute}.
\end{proof}

\begin{lemma}[Expectation decomposition of $\braket{\opR'\Psi}{\opL'\Psi}$; \layermath]
\label{lem:comm-antiherm}
If $\opR$ and $\opL$ are Hermitian, then for every $\ket{\Psi} \in \mathcal{H}$ with $\norm{\Psi} = 1$:
\begin{equation}
  2\,\ImPart\braket{\opR'\Psi}{\opL'\Psi} = \ImPart\braket{\Psi}{\comm{\opR}{\opL}\Psi},
  \qquad
  2\,\RePart\braket{\opR'\Psi}{\opL'\Psi} = \RePart\braket{\Psi}{\{\opR',\opL'\}\Psi}.
  \label{eq:comm-imag}
\end{equation}
Moreover, $\comm{\opR}{\opL}$ is anti-Hermitian ($\comm{\opR}{\opL}^\dagger = -\comm{\opR}{\opL}$),
so $\braket{\Psi}{\comm{\opR}{\opL}\Psi}$ is purely imaginary, while $\{\opR',\opL'\}$ is
Hermitian, so $\braket{\Psi}{\{\opR',\opL'\}\Psi}$ is purely real.
\end{lemma}
\begin{proof}
Because $\opR'$ and $\opL'$ are Hermitian, $\braket{\opR'\Psi}{\opL'\Psi} = \braket{\Psi}{\opR'\opL'\Psi}$
and its complex conjugate is $\braket{\opL'\Psi}{\opR'\Psi} = \braket{\Psi}{\opL'\opR'\Psi}$.
Taking $\frac{1}{2}(\opR'\opL' + \opL'\opR')$ yields the real part (twice the covariance
$\mathrm{Cov}_\Psi(\opR,\opL) = \frac{1}{2}\braket{\Psi}{\{\opR',\opL'\}\Psi}$), while
taking $\frac{1}{2i}(\opR'\opL' - \opL'\opR')$ cancels the scalar shifts $\expect{\opR}{\Psi},
\expect{\opL}{\Psi}$ since identity multiples commute with all operators, leaving
$\comm{\opR'}{\opL'} = \comm{\opR}{\opL}$ and proving both identities in~\eqref{eq:comm-imag}.
Both identities are formally verified as \texttt{centered\_\allowbreak{}comm\_im} and
\texttt{centered\_\allowbreak{}anticomm\_re} in \texttt{RST.Uncertainty.\allowbreak{}Robertson}.
\end{proof}

\subsection{Derivation of the Schr\"odinger and Robertson Bounds}
\label{subsec:unc-derivation}

\begin{theorem}[Schr\"odinger and Robertson Rank-Role Inequalities; \layermath]
\label{thm:rank-role-inequality}
Let $\opR$ and $\opL$ be Hermitian operators on $\mathcal{H} \cong \Cnum^d$ and let
$\ket{\Psi} \in \mathcal{H}$ satisfy $\norm{\Psi} = 1$. Writing the symmetric covariance
term $\mathrm{Cov}_{\Psi}(\opR,\opL) = \frac{1}{2}\braket{\Psi}{\{\opR',\opL'\}\Psi}$, the
dispersions $\Delta R$ and $\Delta L$ obey the \emph{Schr\"odinger uncertainty bound}~\citep{schrodinger1930unscharfeprinzip}:
\begin{equation}
  (\Delta R)^2 (\Delta L)^2
  \;\ge\; \bigl(\mathrm{Cov}_{\Psi}(\opR,\opL)\bigr)^2
  + \frac{1}{4}\bigl\lvert \braket{\Psi}{\comm{\opR}{\opL}\Psi} \bigr\rvert^2,
  \label{eq:rank-role-inequality-sq}
\end{equation}
or equivalently $\Delta R \cdot \Delta L \ge B_S(\Psi)$, where
\begin{equation}
  B_S(\Psi) = \sqrt{\bigl(\mathrm{Cov}_{\Psi}(\opR,\opL)\bigr)^2
  + \frac{1}{4}\bigl\lvert \braket{\Psi}{\comm{\opR}{\opL}\Psi} \bigr\rvert^2}.
  \label{eq:tighter-bound}
\end{equation}
Dropping the non-negative covariance square yields the weaker \emph{Robertson commutator
bound}~\citep{robertson1929uncertainty}:
\begin{equation}
  \Delta R \cdot \Delta L
  \;\ge\; \frac{1}{2}\bigl\lvert \braket{\Psi}{\comm{\opR}{\opL}\Psi} \bigr\rvert
  \;=\; \kappa_R(\Psi).
  \label{eq:rank-role-inequality}
\end{equation}
\end{theorem}
\begin{proof}
Set $\ket{u} = \opR'\ket{\Psi}$ and $\ket{v} = \opL'\ket{\Psi}$. By Cauchy-Schwarz
(\texttt{cauchy\_schwarz\_cpx} in Lean~4; proof in \cref{app:proof-cs}),
\begin{equation}
  (\Delta R)^2 (\Delta L)^2
  = \norm{u}^2 \norm{v}^2
  \;\ge\; \abs{\braket{u}{v}}^2
  = \bigl(\RePart\braket{u}{v}\bigr)^2 + \bigl(\ImPart\braket{u}{v}\bigr)^2.
  \label{eq:cs-step}
\end{equation}
Substituting the anticommutator and commutator identities from \cref{lem:comm-antiherm}
gives
\begin{equation}
  4\,(\Delta R)^2(\Delta L)^2
  \;\ge\; \bigl(\RePart\braket{\Psi}{\{\opR',\opL'\}\Psi}\bigr)^2
  + \bigl(\ImPart\braket{\Psi}{\comm{\opR}{\opL}\Psi}\bigr)^2,
  \label{eq:im-identity}
\end{equation}
which divides by $4$ to establish~\eqref{eq:rank-role-inequality-sq} and~\eqref{eq:tighter-bound}.
Since $(\mathrm{Cov}_\Psi(\opR,\opL))^2 \ge 0$, omitting it proves~\eqref{eq:rank-role-inequality}.
Both~\eqref{eq:im-identity} and its Robertson corollary are machine-checked without division
as \texttt{schrodinger\_\allowbreak{}uncertainty} and \texttt{robertson\_\allowbreak{}uncertainty}.
\end{proof}

\begin{principle}[Situationship constant $\kappaS$; \layerformal{} / \layersatire]
\label{prin:uncertainty}
For fixed non-commuting operators $(\opR, \opL)$ and state $\ket{\Psi}$, define the
state-dependent situationship bound
\begin{equation}
  \kappaS = \kappa_R(\Psi) = \frac{1}{2}\bigl\lvert \ImPart\braket{\Psi}{\comm{\opR}{\opL}\Psi} \bigr\rvert.
  \label{eq:kappa-def}
\end{equation}
Whenever $\kappaS > 0$, $\Delta R \cdot \Delta L \ge B_S(\Psi) \ge \kappaS > 0$: rank
uncertainty and role uncertainty cannot simultaneously vanish.
\end{principle}

\begin{remark}[Why both Robertson and Schr\"odinger matter; \layermath]
\label{rem:covariance-bound}
In v1.0, the anticommutator covariance term was noted only in passing, leaving large slack
when $[\opR,\opL]$ has small expectation alongside strong correlation between $\opR$ and $\opL$.
In Experiment~5 (\cref{sec:simulations}), across $3{,}200$ random operator pairs, Robertson's
median bound for nonzero $\gamma$ is $0.1087$ against a median observed product of $0.4996$,
whereas Schr\"odinger's bound~\eqref{eq:tighter-bound} reaches median $0.4409$, closing most
of the gap between commutator lower bound and actual dispersion product.
\end{remark}

\begin{proposition}[Spectral upper bound on uncertainty products; \layermath]
\label{prop:popoviciu}
If the eigenvalues of $\opR$ lie in $[r_{\min}, r_{\max}]$ and those of $\opL$ lie in
$[\ell_{\min}, \ell_{\max}]$, then for every normalized state $\ket{\Psi}$:
\begin{equation}
  \Delta R \;\le\; \frac{r_{\max} - r_{\min}}{2},
  \qquad
  \Delta L \;\le\; \frac{\ell_{\max} - \ell_{\min}}{2},
  \qquad
  \Delta R \cdot \Delta L \;\le\; \frac{(r_{\max} - r_{\min})(\ell_{\max} - \ell_{\min})}{4}.
  \label{eq:spectral-cap}
\end{equation}
\end{proposition}
\begin{proof}
Shifting $\opR$ by $c = (r_{\max} + r_{\min})/2$ preserves variance, while every eigenvalue
of $\opR - c\opI$ has magnitude at most $(r_{\max} - r_{\min})/2$. Thus
$\var{\opR}{\Psi} = \expect{(\opR - \expect{\opR}{\Psi}\opI)^2}{\Psi}
\le \expect{(\opR - c\opI)^2}{\Psi} \le (r_{\max} - r_{\min})^2 / 4$.
Applying the identical bound to $\opL$ and multiplying gives~\eqref{eq:spectral-cap}.
\end{proof}

\subsection{Minimum-Uncertainty States and Commutative Special Cases}
\label{subsec:unc-tightness}

\begin{proposition}[Minimum-uncertainty saturation criterion; \layermath]
\label{prop:minimum-uncertainty}
Equality holds in Schr\"odinger's bound~\eqref{eq:rank-role-inequality-sq} if and only if
$\opR'\ket{\Psi}$ and $\opL'\ket{\Psi}$ are $\Cnum$-linearly dependent (so Cauchy-Schwarz is
tight). Equality holds in Robertson's bound~\eqref{eq:rank-role-inequality} if and only if
$\opR'\ket{\Psi} = i\lambda \opL'\ket{\Psi}$ for some $\lambda \in \Rnum$ (with zero
covariance $\mathrm{Cov}_\Psi(\opR,\opL) = 0$). In particular, in dimension $d = 2$, every
pure normalized state automatically saturates Schr\"odinger's inequality because the
orthogonal complement of $\ket{\Psi}$ is one-dimensional:
\begin{equation}
  (\Delta R)^2 (\Delta L)^2 = \bigl(\mathrm{Cov}_\Psi(\opR,\opL)\bigr)^2 + \kappa_R(\Psi)^2
  \qquad (d = 2,\ \norm{\Psi} = 1).
  \label{eq:min-unc-eq}
\end{equation}
\end{proposition}
\begin{proof}
By orthogonality $\braket{\Psi}{\opR'\Psi} = \braket{\Psi}{\opL'\Psi} = 0$, both $\opR'\ket{\Psi}$
and $\opL'\ket{\Psi}$ lie in $\mathrm{span}(\ket{\Psi})^\perp$. Equality in Cauchy-Schwarz
occurs iff they are proportional ($\opR'\ket{\Psi} = \alpha\opL'\ket{\Psi}$, or one vanishes);
the real part $\mathrm{Cov}_\Psi$ vanishes iff $\alpha \in i\Rnum$. When $d = 2$,
$\dim(\mathrm{span}(\ket{\Psi})^\perp) = 1$, forcing proportionality for every pure state.
\end{proof}

\subsection{Explicit Rational and Complex Unitary Witnesses}
\label{subsec:unc-worked}

To make the commutator, covariance, and bound tightness completely concrete, we give
two exact rational/Gaussian-rational witnesses derived by unitary rotation and formally
verified in \texttt{RST.Uncertainty.Robertson}, followed by a $4\times 4$ rank-shift example.

\begin{example}[Exact $2\times 2$ witness (\textnormal{\texttt{ex\_2d\_\allowbreak{}schrodinger\_verified}}); \layerformal]
\label{ex:uncertainty-2x2}
Let $D_{R,2} = \diag(1, 2)$, $D_{L,2} = \diag(-1, 1)$, and apply the exact unitary context
rotation $U_2 = \frac{1}{5}\begin{psmallmatrix} 3 & 4i \\ 4i & 3 \end{psmallmatrix}$
($U_2^\dagger U_2 = I_2$, proved as \texttt{U2\_unitary}):
\begin{equation*}
  \opL_2 = U_2^\dagger D_{L,2} U_2 =
  \frac{1}{25}\begin{pmatrix} 7 & -24i \\ 24i & -7 \end{pmatrix},
  \qquad
  \comm{D_{R,2}}{\opL_2} =
  \frac{1}{25}\begin{pmatrix} 0 & 24i \\ 24i & 0 \end{pmatrix}.
\end{equation*}
Evaluate on the unit state $\ket{\Psi_2} = \frac{1}{5}(3, 4)^\top$:
\begin{align*}
  \expect{D_{R,2}}{\Psi_2} &= \frac{41}{25}, &
  \var{D_{R,2}}{\Psi_2} &= \frac{144}{625}, &
  \expect{\opL_2}{\Psi_2} &= -\frac{49}{625}, &
  \var{\opL_2}{\Psi_2} &= \frac{388{,}224}{390{,}625},
\end{align*}
with commutator expectation $\braket{\Psi_2}{\comm{D_{R,2}}{\opL_2}\Psi_2} = \frac{576}{625}i$
and centered anticommutator $\RePart\braket{\Psi_2}{\{\opR',\opL'\}\Psi_2} = -\frac{4{,}032}{15{,}625}$.
Direct multiplication gives
\begin{equation*}
  4\,\var{D_{R,2}}{\Psi_2}\var{\opL_2}{\Psi_2}
  = \frac{223{,}617{,}024}{244{,}140{,}625}
  = \left(\frac{576}{625}\right)^2 + \left(-\frac{4{,}032}{15{,}625}\right)^2,
\end{equation*}
confirming exact Schr\"odinger saturation ($\Delta R \cdot \Delta L = B_S \approx 0.47852 > \kappa_R = 0.46080$)
in $d = 2$ as predicted by \cref{prop:minimum-uncertainty}.
\end{example}

\begin{example}[Exact $3\times 3$ unitary witness with strict slack (\textnormal{\texttt{ex\_3d\_\allowbreak{}schrodinger\_verified}}); \layerformal]
\label{ex:uncertainty-3x3}
Extend $U_2$ onto three partners via $U_3 = U_2 \oplus [1]$, $D_{R,3} = \diag(1, 2, 3)$,
and $D_{L,3} = \diag(-1, 0, 1)$. Then
\begin{equation*}
  \opL_3 = U_3^\dagger D_{L,3} U_3 =
  \frac{1}{25}\begin{pmatrix} -9 & -12i & 0 \\ 12i & -16 & 0 \\ 0 & 0 & 25 \end{pmatrix}.
\end{equation*}
Evaluating on the normalized vector $\ket{\Psi_3} = \bigl(\frac{9}{25}, \frac{12}{25}, \frac{4}{5}\bigr)^\top$
yields $\var{D_{R,3}}{\Psi_3} = \frac{198{,}864}{390{,}625}$, $\var{\opL_3}{\Psi_3} = \frac{155{,}101{,}536}{244{,}140{,}625}$,
$\ImPart\braket{\Psi_3}{\comm{D_{R,3}}{\opL_3}\Psi_3} = \frac{2{,}592}{15{,}625}$, and
$\RePart\braket{\Psi_3}{\{\opR',\opL'\}\Psi_3} = \frac{8{,}966{,}304}{9{,}765{,}625}$. Hence:
\begin{equation*}
  \Delta R \cdot \Delta L \approx 0.56870
  \;>\; B_S(\Psi_3) \approx 0.46651
  \;>\; \kappa_R(\Psi_3) = 0.08294.
\end{equation*}
Here the commutator alone captures less than $15\%$ of the uncertainty product, whereas
including the anticommutator covariance lifts the lower bound to over $82\%$ of $\Delta R \cdot \Delta L$.
\end{example}

\begin{example}[Four-partner cyclic shift operator; \layerformal]
\label{ex:uncertainty-4x4}
Let $d = 4$, $\opR = \diag(1, 2, 3, 4)$, and $\opL = \frac{i}{2}(S - S^\dagger)$ where $S$
is the cyclic shift $S\ket{e_k} = \ket{e_{k+1 \bmod 4}}$ (indices $0,\dots,3$). On the
three-partner uniform state $\ket{\Psi_3'} = \frac{1}{\sqrt{3}}(1, 1, 1, 0)^\top$
(dropping the fourth contingency slot), $\expect{\opR}{\Psi_3'} = 2$,
$\expect{\opL}{\Psi_3'} = 0$, $\Delta R = \sqrt{2/3} \approx 0.8165$, and
$\Delta L = \sqrt{1/6} \approx 0.4082$, while $\mathrm{Cov}_{\Psi_3'}(\opR,\opL) = 0$ and
$\Delta R \cdot \Delta L = \frac{1}{3} = \kappa_R$, yielding exact Robertson saturation.
\end{example}
