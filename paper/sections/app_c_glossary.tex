\section{Glossary}
\label{app:glossary}

\begin{description}[leftmargin=1.8em,style=nextline]
  \item[Amplitudes ($a_i, c_{ij}$)] Complex-valued coefficients whose squared moduli
        $w_i = \abs{a_i}^2$ and $p_{ij} = \abs{c_{ij}}^2$ specify system-level
        primary weights and partner-level role fractions (\cref{sec:framework}).
  \item[Attractiveness-Optionality Conjecture] The falsifiable proposition that
        viable partner count $n(A)$ grows with perceived attractiveness $A$,
        driving higher prevalence of relational superposition (\cref{sec:conjecture}).
  \item[Commutator ($\comm{\opR}{\opL}$)] The operator difference $\opR\opL - \opL\opR$
        measuring whether rank and role queries can be jointly resolved without
        order sensitivity (\cref{sec:uncertainty}).
  \item[Decoherence] System-wide loss of relational coherence in which cross-partner
        verification across a shared social graph drives all active partner weights
        toward zero (\cref{sec:poke,sec:cascade}).
  \item[Dispersion ($\Delta R, \Delta L$)] Standard deviation of outcomes under
        measurement of the rank operator $\opR$ or role operator $\opL$
        (\cref{def:dispersion}).
  \item[Mirror configuration] The gender-inverted configuration in which a male
        focal individual maintains multiple female partners across equivalent
        functional roles (\cref{sec:symmetry}).
  \item[Percolation threshold ($q_{\mathrm{ref}}$)] Branching-process unit-reproduction
        reference density $q_{\mathrm{ref}} = ((n-1)p_{\mathrm{tx}})^{-1}$ benchmarked
        against finite-graph percolation crossovers (\cref{thm:percolation-threshold}).
  \item[Poke ($\poke$)] Any direct question, boundary demand, or scheduling
        confrontation forcing resolution of relational rank (\cref{def:poke}).
  \item[Poke Rule] Projective collapse law $\ket{\Psi} \xrightarrow{\poke} \ket{G_k}$
        with selection probability $\abs{a_k}^2$ (\cref{prin:poke-rule}).
  \item[Rank] Competition ordinal rank $r_i = 1 + \#\{j : w_j > w_i\}$ induced by weight
        vector $\mathbf{w}$ (\cref{def:rank}); unique rank~$1$ fails under
        \textnormal{\texttt{SharedMaximum}} (\cref{prop:nobody-first}).
  \item[Rank-Role Inequality] The Schr\"odinger covariance-inclusive bound
        $\Delta R \cdot \Delta L \ge B_S(\Psi)$ and its Robertson corollary
        $\Delta R \cdot \Delta L \ge \kappa_R = \frac{1}{2}\abs{\ImPart\braket{\Psi}{\comm{\opR}{\opL}\Psi}}$
        (\cref{thm:rank-role-inequality}).
  \item[Relational superposition] Portfolio state with multiple active connections
        (\textnormal{\texttt{DistributedWeights}}, ensuring $w_k < 1$); strong rank
        indeterminacy requires \textnormal{\texttt{SharedMaximum}} or role-scheme rank
        reversal (\cref{def:superposition}).
  \item[Role] Identifiable relational function performed by a partner, represented
        by an orthonormal basis vector $\ket{r_j}$ (\cref{def:role}).
  \item[Silence phase] Latency spike following a poke prior to projective
        selection or cascade (\cref{sec:poke}).
  \item[Situationship constant ($\kappaS$)] Positive lower bound on joint
        rank-role dispersion determined by commutator expectation (\cref{prin:uncertainty}).
\end{description}
